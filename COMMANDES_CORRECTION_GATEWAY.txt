# MÉTHODE 1: SCRIPT AUTOMATIQUE (RECOMMANDÉ)
# ----------------------------------------------------------------------------

# Télécharger le script depuis votre repo GitHub (après l'avoir poussé)
# OU créer le fichier manuellement:

nano reload-gateway-config.sh
# Copier-coller le contenu du fichier reload-gateway-config.sh
# Ctrl+O pour sauvegarder, Ctrl+X pour quitter

# Rendre le script exécutable
chmod +x reload-gateway-config.sh

# Exécuter le script
./reload-gateway-config.sh

# ----------------------------------------------------------------------------
# MÉTHODE 2: COMMANDES MANUELLES (SI VOUS PRÉFÉREZ)
# ----------------------------------------------------------------------------

# Étape 1: Arrêter les services
docker-compose stop gateway-pvvih api-configuration

# Étape 2: Supprimer les conteneurs (IMPORTANT pour forcer le rechargement)
docker-compose rm -f gateway-pvvih api-configuration

# Étape 3: Vérifier qu'ils sont bien supprimés
docker ps -a | grep -E "gateway-pvvih|api-configuration"
# Résultat attendu: aucune ligne (ou seulement des conteneurs "Exited")

# Étape 4: Redémarrer Config Server
docker-compose up -d api-configuration

# Étape 5: Attendre 20 secondes
echo "Attente 20 secondes..."
sleep 20

# Étape 6: Vérifier que Config Server a chargé la bonne config
curl http://localhost:8888/GETWAY_PVVIH/prod | grep -i "USER_API_PVVIH"
# Résultat attendu: vous devez voir "USER_API_PVVIH" (avec underscores)
# Si vous voyez "USER-API-PVVIH" (avec hyphens), la config GitHub n'est pas à jour!

# Étape 7: Redémarrer Gateway
docker-compose up -d gateway-pvvih

# Étape 8: Attendre 30 secondes
echo "Attente 30 secondes..."
sleep 30

# Étape 9: Vérifier les logs du Gateway
docker-compose logs --tail=50 gateway-pvvih

# Étape 10: Vérifier que les routes sont chargées
curl http://localhost:8080/actuator/gateway/routes | jq '.'

# Étape 11: Chercher spécifiquement la route user-api-auth
curl http://localhost:8080/actuator/gateway/routes | jq '.[] | select(.route_id=="user-api-auth")'

# ============================================================================
# NOTES IMPORTANTES
# ============================================================================

# 1. docker-compose restart NE RECHARGE PAS la config depuis GitHub
#    Il faut OBLIGATOIREMENT faire: rm -f puis up -d

# 2. L'ordre de démarrage est important:
#    api-configuration → attendre 20s → gateway-pvvih → attendre 30s

# 3. Les noms de services dans Eureka utilisent des UNDERSCORES:
#    USER_API_PVVIH (pas USER-API-PVVIH)

# 4. Le Gateway doit utiliser lb:// pour la découverte Eureka:
#    uri: lb://USER_API_PVVIH

# 5. Si vous modifiez la config sur GitHub, attendez toujours 20-30 secondes
#    avant de tester (le Config Server met en cache)
