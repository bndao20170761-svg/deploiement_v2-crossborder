  PRÉPARATION
═══════════════════════════════════════════════════════════════════

# Aller dans le répertoire du projet
cd ~/deploiement_v2-crossborder

# Vérifier que le fichier .env existe
cat .env | grep PUBLIC_IP

# Vérifier que Docker fonctionne
docker ps

═══════════════════════════════════════════════════════════════════
  FRONTEND 1: gestion_forum_front (Port 3001)
═══════════════════════════════════════════════════════════════════

# 1. Vérifier le fichier .env du frontend
cat gestion_forum_front/.env

# 2. Builder l'image Docker
docker-compose build gestion-forum-front

# 3. Démarrer le conteneur
docker-compose up -d gestion-forum-front

# 4. Vérifier que ça tourne
docker-compose ps gestion-forum-front

# 5. Voir les logs
docker-compose logs gestion-forum-front

# 6. Tester l'accès
curl -I http://localhost:3001

═══════════════════════════════════════════════════════════════════
  COMMANDES UTILES
═══════════════════════════════════════════════════════════════════

# Arrêter un frontend
docker-compose stop gestion-forum-front

# Redémarrer un frontend
docker-compose restart gestion-forum-front

# Reconstruire un frontend (si vous modifiez le code)
docker-compose build --no-cache gestion-forum-front
docker-compose up -d gestion-forum-front

# Voir les logs en temps réel
docker-compose logs -f gestion-forum-front

# Supprimer un conteneur
docker-compose rm -f gestion-forum-front

# Voir l'utilisation des ressources
docker stats

═══════════════════════════════════════════════════════════════════
  DÉPANNAGE
═══════════════════════════════════════════════════════════════════

# Si le build échoue:
1. Vérifier les logs: docker-compose logs gestion-forum-front
2. Vérifier l'espace disque: df -h
3. Nettoyer Docker: docker system prune -f
4. Rebuilder: docker-compose build --no-cache gestion-forum-front

# Si le conteneur ne démarre pas:
1. Voir les logs: docker-compose logs gestion-forum-front
2. Vérifier le port: sudo netstat -tulpn | grep 3001
3. Redémarrer: docker-compose restart gestion-forum-front

# Si l'accès ne fonctionne pas:
1. Vérifier le pare-feu GCP (ports 3001, 3002, 3003)
2. Vérifier que le conteneur tourne: docker-compose ps
3. Tester depuis la VM: curl http://localhost:3001
