 CORRECTION COMPLÈTE - Port 3003 HTTPS
 À Exécuter Sur le Serveur GCP (100.48.20.109)
=================================================================

# 1. SE CONNECTER AU SERVEUR (si pas déjà connecté)
ssh ec2-user@100.48.20.109

# 2. ALLER DANS LE RÉPERTOIRE
cd ~/deploiement_v2-crossborder

# 3. VÉRIFIER L'ÉTAT ACTUEL
echo "=== État actuel des conteneurs ==="
docker ps | grep -E "(nginx-https|a-user-front)"

echo ""
echo "=== Configuration nginx actuelle ==="
docker exec nginx-https cat /etc/nginx/conf.d/default.conf | grep "listen.*ssl"

# 4. ARRÊTER LES SERVICES CONCERNÉS
echo ""
echo "=== Arrêt des services ==="
docker compose stop a-user-front nginx-https

# 5. SUPPRIMER LES CONTENEURS
echo ""
echo "=== Suppression des conteneurs ==="
docker compose rm -f a-user-front nginx-https

# 6. REDÉMARRER AVEC LA NOUVELLE CONFIGURATION
echo ""
echo "=== Redémarrage ==="
docker compose up -d a-user-front nginx-https

# 7. ATTENDRE LE DÉMARRAGE
echo ""
echo "=== Attente du démarrage (10 secondes) ==="
sleep 10

# 8. VÉRIFIER LES CONTENEURS
echo ""
echo "=== Vérification des conteneurs ==="
docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}" | grep -E "(nginx-https|a-user-front)"

# 9. VÉRIFIER LA CONFIGURATION NGINX
echo ""
echo "=== Configuration nginx (doit montrer 'listen 3003 ssl') ==="
docker exec nginx-https cat /etc/nginx/conf.d/default.conf | grep "listen.*3003"

# 10. TESTER HTTP DIRECT (port 3013)
echo ""
echo "=== Test HTTP direct (port 3013) ==="
curl -I http://localhost:3013/

# 11. TESTER HTTPS VIA NGINX (port 3003)
echo ""
echo "=== Test HTTPS via nginx (port 3003) ==="
curl -k -I https://localhost:3003/

# 12. VÉRIFIER LES LOGS EN CAS DE PROBLÈME
echo ""
echo "=== Logs nginx-https (dernières lignes) ==="
docker logs nginx-https --tail 20

echo ""
echo "=== Logs a-user-front (dernières lignes) ==="
docker logs a-user-front --tail 20
