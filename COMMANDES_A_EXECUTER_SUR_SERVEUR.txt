# 1. SE CONNECTER AU SERVEUR
# ----------------------------------------
ssh ec2-user@100.48.20.109


# 2. ALLER DANS LE RÉPERTOIRE DU PROJET
# ----------------------------------------
cd ~/deploiement_v2-crossborder


# 3. VÉRIFIER L'ÉTAT ACTUEL
# ----------------------------------------
docker ps | grep -E "(nginx-https|gateway-pvvih)"
docker logs nginx-https --tail 20


# 4. TESTER LE PROBLÈME
# ----------------------------------------
# Test port 8080 SSL (devrait échouer ou retourner du HTML)
curl -k https://localhost:8080/actuator/health

# Test Gateway direct sans SSL (devrait fonctionner)
curl http://gateway-pvvih:8080/actuator/health


# 5. CORRIGER LA CONFIGURATION
# ----------------------------------------
# Backup de la config actuelle
cp nginx-https.conf nginx-https.conf.backup.$(date +%Y%m%d_%H%M%S)

# 6. REDÉMARRER NGINX-HTTPS
# ----------------------------------------
docker compose restart nginx-https

sleep 5

# Vérifier que c'est healthy
docker ps | grep nginx-https


# 7. TESTER LA CORRECTION
# ----------------------------------------
# Test port 8080 SSL (devrait maintenant retourner du JSON!)
curl -k https://localhost:8080/actuator/health

# Devrait afficher quelque chose comme :
# {"status":"UP","components":{"ping":{"status":"UP"}}}


# 8. TESTER TOUS LES PORTS
# ----------------------------------------
echo "Test port 443:"
curl -k -I https://localhost:443/

echo "Test port 3001 (a-reference-front):"
curl -k -I https://localhost:3001/

echo "Test port 3003 (a-user-front):"
curl -k -I https://localhost:3003/

echo "Test port 8080 (Gateway):"
curl -k https://localhost:8080/actuator/health


# 9. VÉRIFIER LES LOGS EN CAS DE PROBLÈME
# ----------------------------------------

docker compose restart

# Vérifier les certificats SSL
docker exec nginx-https ls -la /etc/nginx/ssl/

# Vérifier la configuration chargée
docker exec nginx-https cat /etc/nginx/conf.d/default.conf | grep -A 10 "listen 8080"
