═══════════════════════════════════════════════════════════════
ÉTAPE 1 : Récupérer les changements depuis GitHub
═══════════════════════════════════════════════════════════════

cd ~/deploiement_v2-crossborder
git pull origin main

═══════════════════════════════════════════════════════════════
ÉTAPE 2 : Arrêter les services backend
═══════════════════════════════════════════════════════════════

docker-compose stop gateway-pvvih gestion-user gestion-reference gestion-patient forum-pvvih

═══════════════════════════════════════════════════════════════
ÉTAPE 3 : Supprimer les conteneurs (pour recharger les variables)
═══════════════════════════════════════════════════════════════

docker-compose rm -f gateway-pvvih gestion-user gestion-reference gestion-patient forum-pvvih

═══════════════════════════════════════════════════════════════
ÉTAPE 4 : Reconstruire les images
═══════════════════════════════════════════════════════════════

docker-compose build gateway-pvvih gestion-user gestion-reference gestion-patient forum-pvvih

═══════════════════════════════════════════════════════════════
ÉTAPE 5 : Redémarrer les services
═══════════════════════════════════════════════════════════════

docker-compose up -d gateway-pvvih gestion-user gestion-reference gestion-patient forum-pvvih

═══════════════════════════════════════════════════════════════
ÉTAPE 6 : Voir les logs en temps réel
═══════════════════════════════════════════════════════════════

docker-compose logs -f gestion-user gestion-reference

# Pour arrêter les logs : Ctrl+C

═══════════════════════════════════════════════════════════════
ÉTAPE 7 : Vérifier que les services ont démarré
═══════════════════════════════════════════════════════════════

docker-compose logs gestion-user | grep -i "started"
docker-compose logs gestion-reference | grep -i "started"
docker-compose logs gateway-pvvih | grep -i "started"

═══════════════════════════════════════════════════════════════
COMMANDES UTILES
═══════════════════════════════════════════════════════════════

docker ps
