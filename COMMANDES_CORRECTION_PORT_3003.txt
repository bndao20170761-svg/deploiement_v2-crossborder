# ==========================================================
# CORRECTION DÉFINITIVE: Port 3003 HTTPS
# ==========================================================

# PROBLÈME:
# - Le port 3003 était utilisé par DEUX services en même temps:
#   1. a-user-front (HTTP direct depuis le conteneur)
#   2. nginx-https (HTTPS avec SSL)
# - Résultat: ERR_SSL_PROTOCOL_ERROR

# SOLUTION:
# - a-user-front utilise maintenant le port 3013 (HTTP direct)
# - nginx-https utilise le port 3003 (HTTPS avec SSL)

# ==========================================================
# COMMANDES À EXÉCUTER SUR LE SERVEUR
# ==========================================================

# 1. Arrêter et supprimer les conteneurs concernés
docker compose stop a-user-front nginx-https
docker compose rm -f a-user-front nginx-https

# 2. Redémarrer avec la nouvelle configuration
docker compose up -d a-user-front nginx-https

# 3. Vérifier que les services sont démarrés
docker ps | grep -E "(a-user-front|nginx-https)"

# 4. Vérifier la configuration SSL du port 3003
docker exec nginx-https cat /etc/nginx/conf.d/default.conf | grep "listen.*3003"

# 5. Tester depuis le serveur
echo "Test HTTP direct (port 3013):"
curl -I http://localhost:3013/

echo ""
echo "Test HTTPS via nginx (port 3003):"
curl -k -I https://localhost:3003/
