==========================================================================
🔧 COMMANDES À EXÉCUTER SUR LE SERVEUR GCP - CORRECTION NGINX
==========================================================================

PROBLÈME ACTUEL :
- ❌ Port 8080 ne répond pas correctement (ERR_SSL_PROTOCOL_ERROR)
- ❌ Login impossible sur tous les frontends
- ❌ nginx-https.conf sur le serveur est obsolète

SOLUTION : Mettre à jour nginx-https.conf sur le serveur GCP

==========================================================================

📋 ÉTAPE 1 : SE CONNECTER AU SERVEUR
----------------------------------------
ssh ec2-user@100.48.20.109


📋 ÉTAPE 2 : ALLER DANS LE RÉPERTOIRE DU PROJET
----------------------------------------
cd ~/deploiement_v2-crossborder


📋 ÉTAPE 3 : SAUVEGARDER L'ANCIENNE CONFIGURATION
----------------------------------------
cp nginx-https.conf nginx-https.conf.backup.$(date +%Y%m%d_%H%M%S)


📋 ÉTAPE 4 : TÉLÉCHARGER LA NOUVELLE CONFIGURATION
----------------------------------------
# Méthode A : Si le fichier CORRIGER_NGINX_HTTPS_SUR_SERVEUR.sh existe
chmod +x CORRIGER_NGINX_HTTPS_SUR_SERVEUR.sh
./CORRIGER_NGINX_HTTPS_SUR_SERVEUR.sh

# OU Méthode B : Modification manuelle
nano nginx-https.conf

# Puis ajouter à la fin du fichier (avant la dernière accolade) :

# Serveur pour API Gateway sur port 8080 (CRITIQUE)
server {
    listen 8080 ssl;
    server_name _;

    ssl_certificate /etc/nginx/ssl/nginx-selfsigned.crt;
    ssl_certificate_key /etc/nginx/ssl/nginx-selfsigned.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    client_max_body_size 100M;
    proxy_connect_timeout 600;
    proxy_send_timeout 600;
    proxy_read_timeout 600;

    location / {
        proxy_pass http://gateway-pvvih:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
}

# Sauvegarder : Ctrl+X, puis Y, puis Entrée


📋 ÉTAPE 5 : REDÉMARRER NGINX-HTTPS
----------------------------------------
docker compose restart nginx-https


📋 ÉTAPE 6 : ATTENDRE 10 SECONDES
----------------------------------------
sleep 10


📋 ÉTAPE 7 : VÉRIFIER QUE NGINX A DÉMARRÉ
----------------------------------------
docker ps | grep nginx-https
# Devrait afficher "Up X seconds (healthy)"


📋 ÉTAPE 8 : TESTER LE PORT 8080
----------------------------------------
curl -k https://localhost:8080/actuator/health

# Devrait afficher du JSON comme :
# {"status":"UP"}

# Si erreur ou HTML → Le problème persiste


📋 ÉTAPE 9 : TESTER LES AUTRES PORTS
----------------------------------------
echo "Test port 3001:"
curl -k -s -o /dev/null -w "%{http_code}\n" https://localhost:3001/

echo "Test port 3003:"
curl -k -s -o /dev/null -w "%{http_code}\n" https://localhost:3003/

# Devrait afficher : 200 pour chaque


📋 ÉTAPE 10 : VÉRIFIER LES LOGS EN CAS DE PROBLÈME
----------------------------------------
docker logs nginx-https --tail 50
